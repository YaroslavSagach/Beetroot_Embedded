//const int MOTOR_PIN = TX;
#include <Arduino.h>

const int POT_PIN = 1; // Пін потенціометра 

void setup() {
  Serial0.begin(9600);
  pinMode(POT_PIN, INPUT);
}

void loop() {
  // 1. Читаємо значення з потенціометра
  int adcValue = analogRead(POT_PIN);

  // 2. Визначаємо, який байт відправляти залежно від положення
  uint8_t pwmByte;

  // Розподіляємо значення (для класичного АЦП 0...1023)
  // Якщо у вас ESP32 (0...4095), замініть діапазони відповідно
  if (adcValue < 170) {
    pwmByte = 0x00; // ~10% ШІМ (мінімальне заповнення)
  } else if (adcValue < 340) {
    pwmByte = 0x01; // ~20% ШІМ 
  } else if (adcValue < 510) {
    pwmByte = 0x0F; // ~50% ШІМ (низька частота, широкий меандр)
  } else if (adcValue < 680) {
    pwmByte = 0x55; // ~50% ШІМ (висока частота, часті короткі імпульси)
  } else if (adcValue < 850) {
    pwmByte = 0x7F; // ~80% ШІМ
  } else {
    pwmByte = 0xFF; // ~90% ШІМ (максимальне заповнення)
  }

  // 3. Безперервно відправляємо обраний байт в UART для формування сигналу
  Serial0.write(pwmByte);
  
  // Чекаємо завершення відправки байта, щоб не переповнювати буфер
  // і тримати ШІМ-сигнал стабільним та безперервним
  Serial0.flush(); 
}
